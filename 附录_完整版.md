def __get_chain(self):
    """获取最终的执行链"""
    def format_document(docs: list[Document]):
        return "\n\n".join([d.page_content for d in docs])

    def format_for_prompt_template(value):
        if isinstance(value, dict) and "history" in value:
            return value["history"]
        return value

    system_prompt = (
        "你是一个专业的临床决策支持助手，请基于以下参考信息"
        "准确、简洁地回答用户的问题。\n\n"
        "参考信息：\n{context}\n\n"
        "对话历史：\n{history}\n\n"
        "用户问题：{input}"
    )
    prompt = ChatPromptTemplate.from_template(system_prompt)

    retriever = self.vector_service.get_retriever()
    chain = (
        RunnablePassthrough.assign(
            context=lambda x: format_document(
                retriever.invoke(x["input"])
            )
        )
        | prompt
        | self.llm
        | StrOutputParser()
    )

    chain_with_history = RunnableWithMessageHistory(
        chain,
        get_history,
        input_messages_key="input",
        history_messages_key="history",
    )
    return chain_with_history

def extract_text_from_uploaded_file(uploaded_file) -> tuple[str, str]:
    """从上传文件中提取文本内容"""
    file_bytes = uploaded_file.file.read()
    filename = uploaded_file.filename.lower()
    if filename.endswith(".pdf"):
        return _extract_text_from_pdf(file_bytes), "pdf"
    elif filename.endswith(".docx"):
        return _extract_text_from_docx(file_bytes), "docx"
    else:
        return file_bytes.decode("utf-8", errors="ignore"), "txt"


def get_string_md5(input_str: str, encoding='utf-8'):
    return hashlib.md5(input_str.encode(encoding)).hexdigest()


class KnowledgeBaseService(object):
    def upload_by_str(self, data: str, filename, operator: str = ""):

        )
        docs = text_splitter.create_documents([data])
        now = datetime.now().strftime("%Y-%m-%d %H:%M:%S")
        for doc in docs:
            doc.metadata = {
                "source": filename, "create_time": now, "operator": operator,
            }
        self.vector_store.add_documents(docs)
        return f"成功上传 {len(docs)} 个文本块"

def extract_relevant_line(query: str, full_text: str) -> str:
    lines = full_text.split("\n")
    query_chars = set(query)
    best_line, best_score = "", 0
    for line in lines:
        line_stripped = line.strip()
        if not line_stripped:
            continue
        line_chars = set(line_stripped)
        common = query_chars & line_chars
        score = len(common) / max(len(query_chars), len(line_chars))
        if score > best_score:
            best_score = score
            best_line = line_stripped
    return best_line if best_line else full_text[:200]

class FileChatMessageHistory(BaseChatMessageHistory):
    def __init__(self, session_id: str, storage_path: str = None):
        self.session_id = session_id
        self.storage_path = storage_path or db_path

    def add_messages(self, messages: Sequence[BaseMessage]) -> None:
        conn = get_connection()
        try:
            now = datetime.now().strftime("%Y-%m-%d %H:%M:%S")
            for msg in messages:
                role = "human" if isinstance(msg, HumanMessage) else "ai"
                conn.execute(
                    "INSERT INTO chat_history (session_id, type, content, create_time) VALUES (?, ?, ?, ?)",
                    (self.session_id, role, msg.content, now),
                )
            conn.commit()
        finally:
            conn.close()

    @property
    def messages(self) -> List[BaseMessage]:
        conn = get_connection()
        try:
            rows = conn.execute(
                "SELECT type, content FROM chat_history WHERE session_id = ? ORDER BY id",
                (self.session_id,),
            ).fetchall()
            result = []
            for row in rows:
                if row["type"] == "human":
                    result.append(HumanMessage(content=row["content"]))
                else:
                    result.append(AIMessage(content=row["content"]))
            return result
        finally:
            conn.close()

def register(work_id: str, password: str) -> str:
    """注册新用户"""
    conn = get_connection()
    try:
        existing = conn.execute(
            "SELECT id FROM users WHERE work_id = ?", (work_id,)
        ).fetchone()
        if existing:
            return "工号已存在"
        hashed = hashlib.sha256(password.encode()).hexdigest()
        conn.execute(
            "INSERT INTO users (work_id, password) VALUES (?, ?)",
            (work_id, hashed),
        )
        conn.commit()
        return "注册成功"
    finally:
        conn.close()


def login(work_id: str, password: str) -> str:
    """用户登录"""
    conn = get_connection()
    try:
        row = conn.execute(
            "SELECT password FROM users WHERE work_id = ?", (work_id,)
        ).fetchone()
        if row is None:
            return "工号不存在"
        hashed = hashlib.sha256(password.encode()).hexdigest()
        return "登录成功" if row["password"] == hashed else "密码错误"
    finally:
        conn.close()

def search_records(keyword: str) -> list[dict]:
    """根据关键词搜索诊断记录"""
    conn = get_connection()
    try:
        keyword = keyword.strip()
        if not keyword:
            rows = conn.execute(
                "SELECT * FROM diagnosis_records ORDER BY create_time DESC"
            ).fetchall()
            return [dict(row) for row in rows]
        like_pattern = f"%{keyword}%"
        rows = conn.execute(
            """SELECT * FROM diagnosis_records WHERE patient_name LIKE ? OR id_number LIKE ?
               OR symptoms LIKE ? OR diagnosis_result LIKE ? ORDER BY create_time DESC""",
            (like_pattern, like_pattern, like_pattern, like_pattern),
        ).fetchall()
        return [dict(row) for row in rows]
    finally:
        conn.close()

    return "\n".join(lines)

def get_logs(page: int = 1, page_size: int = 50, action_type: str = None, keyword: str = None) -> dict:
    """分页查询操作日志"""
    conn = get_connection()
    try:
        conditions, params = [], []
        if action_type:
            conditions.append("action_type = ?")
            params.append(action_type)
        if keyword:
            conditions.append("(detail LIKE ? OR operator LIKE ?)")
            like_pattern = f"%{keyword}%"
            params.extend([like_pattern, like_pattern])
        where = (" WHERE " + " AND ".join(conditions)) if conditions else ""
        total = conn.execute(f"SELECT COUNT(*) FROM operation_logs{where}", params).fetchone()[0]
        offset = (page - 1) * page_size
        rows = conn.execute(
            f"SELECT * FROM operation_logs{where} ORDER BY id DESC LIMIT ? OFFSET ?",
            params + [page_size, offset],
        ).fetchall()
        return {"total": total, "page": page, "page_size": page_size, "records": [dict(r) for r in rows]}
    finally:
        conn.close()
